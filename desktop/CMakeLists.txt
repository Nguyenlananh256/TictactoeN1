cmake_minimum_required(VERSION 3.16)
project(CaroArena VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CARO_BUILD_GUI "Build the Qt desktop application" ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(caro_rules_tests tests/rules_test.cpp)
    target_include_directories(caro_rules_tests PRIVATE src)
    add_test(NAME caro_rules COMMAND caro_rules_tests)
endif()
if(CARO_BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets Concurrent)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets Concurrent)
    set(CMAKE_AUTORCC ON)
    add_executable(CaroArena WIN32 src/main.cpp src/window.cpp src/window.h src/board_widget.cpp src/board_widget.h src/game.h src/ai.h assets/fonts.qrc)
    target_link_libraries(CaroArena PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT CaroArena)
    if(WIN32)
        get_target_property(_qt_core_dll Qt${QT_VERSION_MAJOR}::Core IMPORTED_LOCATION_RELEASE)
        get_filename_component(_qt_bin "${_qt_core_dll}" DIRECTORY)
        find_program(CARO_WINDEPLOYQT windeployqt HINTS "${_qt_bin}" NO_DEFAULT_PATH)
        option(CARO_DEPLOY_RUNTIME "Copy Qt runtime DLLs next to the executable" ON)
        if(CARO_DEPLOY_RUNTIME AND CARO_WINDEPLOYQT)
            add_custom_command(TARGET CaroArena POST_BUILD
                COMMAND "${CARO_WINDEPLOYQT}" "$<IF:$<CONFIG:Debug>,--debug,--release>"
                    --no-translations --no-compiler-runtime "$<TARGET_FILE:CaroArena>"
                COMMENT "Deploying Qt runtime for Caro Arena"
                VERBATIM)
        endif()
    endif()
    if(MSVC)
        target_compile_options(CaroArena PRIVATE /utf-8 /W4)
    else()
        target_compile_options(CaroArena PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS CaroArena RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_executable(caro_ui_tests tests/ui_test.cpp src/window.cpp src/board_widget.cpp)
        target_include_directories(caro_ui_tests PRIVATE src)
        target_link_libraries(caro_ui_tests PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
        if(WIN32)
            foreach(_plugin qoffscreen qoffscreend)
                if(EXISTS "${_qt_bin}/../plugins/platforms/${_plugin}.dll")
                    add_custom_command(TARGET caro_ui_tests POST_BUILD
                        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:caro_ui_tests>/platforms"
                        COMMAND ${CMAKE_COMMAND} -E copy_if_different
                            "${_qt_bin}/../plugins/platforms/${_plugin}.dll"
                            "$<TARGET_FILE_DIR:caro_ui_tests>/platforms/"
                        VERBATIM)
                endif()
            endforeach()
        endif()
        if(MSVC)
            target_compile_options(caro_ui_tests PRIVATE /utf-8)
        endif()
        add_test(NAME caro_ui COMMAND caro_ui_tests -platform offscreen)
    endif()
endif()
